\documentclass{article}
\usepackage{pseudocode}

\pagestyle{empty}
\begin{document}
	\begin{pseudocode}{CelsiusToFahrenheit}{c}
		f \GETS {9c/5} + 32 \\
		\RETURN{f}
	\end{pseudocode}
	\\
	\begin{pseudocode}{PythagoreanTheorem}{a, b}
		f \GETS \sqrt{a^2 + b^2} \\
		\RETURN{f}
	\end{pseudocode}
\end{document}
